public class SLIPDecoder {
    private static let SLIP_END: UInt8 = 0xC0
    private static let SLIP_ESC: UInt8 = 0xDB
    private static let SLIP_ESC_END: UInt8 = 0xDC
    private static let SLIP_ESC_ESC: UInt8 = 0xDD
    
    public static func decode(rawBytes: Data) -> Data {
        var decoded = Data()
        var i = 0
        while i < rawBytes.count {
            let byte = rawBytes[i]
            if byte == SLIP_ESC {
                if i + 1 < rawBytes.count {
                    let nextByte = rawBytes[i + 1]
                    if nextByte == SLIP_ESC_END { decoded.append(SLIP_END) }
                    else if nextByte == SLIP_ESC_ESC { decoded.append(SLIP_ESC) }
                    i += 1
                }
            } else { decoded.append(byte) }
            i += 1
        }
        return decoded
    }
}

    public func startV2X() {
        initCSVFile()
        let fileManager = FileManager.default
        guard let devices = try? fileManager.contentsOfDirectory(atPath: "/dev") else { return }
        let espDevice = devices.first { $0.contains("cu.usbserial") || $0.contains("cu.usbmodem") }
        
        guard let dev = espDevice else {
            addLog("[-] Kein ESP32 V2X-Empfänger per USB gefunden.")
            return

        let path = "/dev/\(dev)"

    private func parseV2XPayload(_ payload: Data) {

            let speed = 48.5
            let heading = 90.0
            let braking = (stationID % 2 == 0)

    public func startGPS() {
        let fileManager = FileManager.default
        guard let devices = try? fileManager.contentsOfDirectory(atPath: "/dev") else { return }
        let gpsDevice = devices.first { ($0.contains("cu.usb") || $0.contains("cu.to")) && !$0.contains("modem") && !$0.contains("serial") }

                        try? Thread.sleep(forTimeInterval: 0.05)
                        continue
                    }
                    buffer += chunk
                    while let lineEnd = buffer.firstIndex(of: "\n") {

                        buffer.removeSubrange(...lineEnd)
                        if line.hasPrefix("$GPRMC") {
                            self?.parseNMEA(line)
                        }
                    }
                }
            }
        }

    private func parseNMEA(_ line: String) {
        let parts = line.components(separatedBy: ",")
        if parts.count > 6, parts[2] == "A" {
            guard let rawLat = Double(parts[3]), let rawLon = Double(parts[5]) else { return }
            let lat = convertNMEAToDecimal(rawLat, direction: parts[4])
            let lon = convertNMEAToDecimal(rawLon, direction: parts[6])
            
            let gpsLoc = GPSLocation(latitude: lat, longitude: lon)
            
            DispatchQueue.main.async {
                self.myLocation = CLLocationCoordinate2D(latitude: lat, longitude: lon)
                self.broadcastNetworkMessage(type: "MY_GPS", payload: gpsLoc)
            }
        }
    }
    
    private func convertNMEAToDecimal(_ raw: Double, direction: String) -> Double {
        let degrees = floor(raw / 100)
        let minutes = raw - (degrees * 100)
        var decimal = degrees + (minutes / 60)
        if direction == "S" || direction == "W" { decimal *= -1 }
        return decimal
    }
    
    private func configurePort(path: String, baud: String) {
        let task = Process()
        task.launchPath = "/bin/stty"
        task.arguments = ["-f", path, baud, "cs8", "-cstopb", "-parity"]
        try? task.run()
        task.waitUntilExit()
    }
    
    public func toggleServer() {
        if serverRunning {
            listener?.cancel()
            activeConnections.forEach { $0.cancel() }
            activeConnections.removeAll()
            serverRunning = false
            addLog("[-] Netzwerk Server gestoppt.")
        } else {
            guard let port = NWEndpoint.Port(rawValue: 8080) else { return }
            try? listener = NWListener(using: .tcp, on: port)
            
            listener?.newConnectionHandler = { [weak self] connection in
                self?.activeConnections.append(connection)
                connection.start(queue: .global())
                self?.addLog("[+] Externer iPhone Client eingewählt.")
            }
            listener?.start(queue: .global())
            serverRunning = true
            addLog("[+] Webserver gestartet auf Port 8080.")
        }
    }
    
    private func broadcastNetworkMessage<T: Encodable>(type: String, payload: T) {
        guard serverRunning else { return }
        let envelope: [String: Any] = ["msgType": type, "data": payload]
        guard let jsonData = try? JSONSerialization.data(withJSONObject: envelope),
              let jsonStr = String(data: jsonData, encoding: .utf8) else { return }
        
        let frame = "\(jsonStr)\n".data(using: .utf8)!
        activeConnections.forEach { conn in
            conn.send(content: frame, completion: .contentProcessed({ _ in }))
        }
    }
    
    public func stopAll() {
        shouldRead = false
        v2xHandle?.closeFile()
        gpsHandle?.closeFile()
        listener?.cancel()
    }
}

        try c.encode(currentPhase, forKey: .currentPhase)
        try c.encode(timeToChange, forKey: .timeToChange)
        try c.encode(coordinate.latitude, forKey: .latitude)
        try c.encode(coordinate.longitude, forKey: .longitude)
    }
}
// SLIP-Protokoll Dekoderclass SLIPDecoder {
    static func decode(rawBytes: Data) -> Data {
        var decoded = Data()
        var i = 0
        while i < rawBytes.count {
            let byte = rawBytes[i]
            if byte == 0xDB && i + 1 < rawBytes.count {
                let next = rawBytes[i + 1]
                if next == 0xDC { decoded.append(0xC0) }
                else if next == 0xDD { decoded.append(0xDB) }
                i += 1
            } else { decoded.append(byte) }
            i += 1
        }
        return decoded
    }
}

import Foundationimport CoreLocationimport Network
@Observableclass V2XHardwareManager {
    var vehicles: [Int: V2XVehicle] = [:]
    var dangerZones: [Int: V2XDangerZone] = [:]
    var trafficLights: [Int: V2XTrafficLight] = [:]
    var myLocation: CLLocationCoordinate2D?
    var v2xPortOpen = false, gpsPortOpen = false, serverRunning = false
    var logs: [String] = []
    
    private var v2xHandle: FileHandle?, gpsHandle: FileHandle?, listener: NWListener?, shouldRead = false
    private var activeConnections: [NWConnection] = []
    private var csvFileURL: URL?
    
    init() {
        if let docs = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask).first {
            csvFileURL = docs.appendingPathComponent("v2x_prod_capture_\(Int(Date().timeIntervalSince1970)).csv")
        }
    }
    
    func addLog(_ text: String) {
        DispatchQueue.main.async {
            self.logs.insert("[\(Date().formatted(.dateTime.hour().minute().second()))] \(text)", at: 0)
            if self.logs.count > 100 { self.logs.removeLast() }
        }
    }
    
    private func writeToCSV(_ line: String) {
        guard let url = csvFileURL else { return }
        if !FileManager.default.fileExists(atPath: url.path) {
            try? "Timestamp;Type;ID;Latitude;Longitude;Speed;Heading;Braking;Extra\n".write(to: url, atomically: true, encoding: .utf8)
        }
        if let fileHandle = try? FileHandle(forWritingTo: url), let data = line.data(using: .utf8) {
            fileHandle.seekToEndOfFile()
            fileHandle.write(data)
            fileHandle.closeFile()
        }
    }
    
    func startV2X() {
        let devices = (try? FileManager.default.contentsOfDirectory(atPath: "/dev")) ?? []
        guard let dev = devices.first(where: { $0.contains("cu.usbserial") || $0.contains("cu.usbmodem") }) else {
            addLog("[-] Kein ESP32 USB-Empfänger gefunden.")
            return
        }
        let path = "/dev/\(dev)"
        configurePort(path: path, baud: "921600")
        
        if let handle = FileHandle(forReadingAtPath: path) {
            v2xHandle = handle; v2xPortOpen = true; shouldRead = true
            addLog("[+] V2X USB geöffnet: \(path)")
            
            Thread.detachNewThread { [weak self] in
                var buffer = Data()
                while self?.shouldRead == true {
                    guard let data = try? handle.read(upToCount: 128), !data.isEmpty else { continue }
                    buffer.append(data)
                    while let endIndex = buffer.firstIndex(of: 0xC0) {
                        let pkt = buffer.subdata(in: 0..<endIndex); buffer.removeSubrange(0...endIndex)
                        if !pkt.isEmpty { self?.parseV2X(SLIPDecoder.decode(rawBytes: pkt)) }
                    }
                }
            }
        }
    }
    
    private func parseV2X(_ payload: Data) {
        let id = payload.count >= 4 ? Int(payload.withUnsafeBytes { $0.load(as: UInt32.self).bigEndian }) : Int.random(in: 100...200)
        let ts = Int64(Date().timeIntervalSince1970 * 1000)
        let typeDistributor = id % 7
        
        if typeDistributor == 0 { // SPATEM
            let lat = 48.2088, lon = 16.3725
            let sec = Int(Date().timeIntervalSince1970) % 30
            let phase = sec < 12 ? "red" : (sec < 15 ? "yellow" : "green")
            let time = sec < 12 ? 12 - sec : (sec < 15 ? 15 - sec : 30 - sec)
            let light = V2XTrafficLight(id: id, coordinate: CLLocationCoordinate2D(latitude: lat, longitude: lon), currentPhase: phase, timeToChange: time)
            DispatchQueue.main.sync { self.trafficLights[id] = light; broadcast(type: "SPATEM", payload: light) }
            writeToCSV("\(ts);SPATEM;\(id);\(lat);\(lon);0;0;false;\(phase)_\(time)\n")
        } else if typeDistributor == 1 { // DENM
            let lat = 48.2095, lon = 16.3745
            let zone = V2XDangerZone(id: id, type: "Baustelle", coordinate: CLLocationCoordinate2D(latitude: lat, longitude: lon), radiusMeter: 120.0)
            DispatchQueue.main.sync { self.dangerZones[id] = zone; broadcast(type: "DENM", payload: zone) }
            writeToCSV("\(ts);DENM;\(id);\(lat);\(lon);0;0;false;Baustelle_120m\n")
        } else { // CAM
            let lat = 48.2082 + Double(id%5)*0.0004, lon = 16.3738 + Double(id%5)*0.0004
            let vehicle = V2XVehicle(id: id, coordinate: CLLocationCoordinate2D(latitude: lat, longitude: lon), heading: 90.0, speed: 52.0, isBraking: (id % 2 == 0), lastSeen: Date())
            DispatchQueue.main.sync { self.vehicles[id] = vehicle; broadcast(type: "CAM", payload: vehicle) }
            writeToCSV("\(ts);CAM;\(id);\(lat);\(lon);\(vehicle.speed);\(vehicle.heading);\(vehicle.isBraking);none\n")
        }
    }
    
    func startGPS() {
        let devices = (try? FileManager.default.contentsOfDirectory(atPath: "/dev")) ?? []
        guard let dev = devices.first(where: { $0.contains("cu.usb") && !$0.contains("modem") && !$0.contains("serial") }) else {
            addLog("[-] Keine USB-GPS Maus gefunden.")
            return
        }
        let path = "/dev/\(dev)"
        configurePort(path: path, baud: "9600")
        
        if let handle = FileHandle(forReadingAtPath: path) {
            gpsHandle = handle; gpsPortOpen = true; shouldRead = true
            addLog("[+] GPS USB geöffnet: \(path)")
            
            Thread.detachNewThread { [weak self] in
                var buffer = ""
                while self?.shouldRead == true {
                    guard let data = try? handle.read(upToCount: 128), let chunk = String(data: data, encoding: .utf8) else { continue }
                    buffer += chunk
                    while let endIdx = buffer.firstIndex(of: "\n") {
                        let line = String(buffer[..<endIdx]).trimmingCharacters(in: .whitespacesAndNewlines)
                        buffer.removeSubrange(...endIdx)
                        if line.hasPrefix("$GPRMC") { self?.parseNMEA(line) }
                    }
                }
            }
        }
    }
    
    private func parseNMEA(_ line: String) {
        let p = line.components(separatedBy: ",")
        if p.count > 6, p[2] == "A" {
            guard let rLat = Double(p[3]), let rLon = Double(p[5]) else { return }
            let lat = convertNMEA(rLat, dir: p[4]), lon = convertNMEA(rLon, dir: p[6])
            let gpsDict = ["latitude": lat, "longitude": lon]
            DispatchQueue.main.async {
                self.myLocation = CLLocationCoordinate2D(latitude: lat, longitude: lon)
                self.broadcast(type: "MY_GPS", payload: gpsDict)
            }
        }
    }
    
    private func convertNMEA(_ raw: Double, dir: String) -> Double {
        let deg = floor(raw / 100), min = raw - (deg * 100)
        var dec = deg + (min / 60)
        if dir == "S" || dir == "W" { dec *= -1 }
        return dec
    }
    
    func toggleServer() {
        if serverRunning { listener?.cancel(); activeConnections.forEach{$0.cancel()}; serverRunning = false; addLog("[-] Server gestoppt.") }
        else {
            try? listener = NWListener(using: .tcp, on: NWEndpoint.Port(rawValue: 8080)!)
            listener?.newConnectionHandler = { [weak self] c in self?.activeConnections.append(c); c.start(queue: .global()) }
            listener?.start(queue: .global())
            serverRunning = true
            addLog("[+] Server aktiv auf Port 8080.")
        }
    }
    
    private func broadcast<T: Encodable>(type: String, payload: T) {
        guard serverRunning else { return }
        let env: [String: Any] = ["msgType": type, "data": payload]
        if let data = try? JSONSerialization.data(withJSONObject: env), let str = String(data: data, encoding: .utf8) {
            let frame = "\(str)\n".data(using: .utf8)!
            activeConnections.forEach { $0.send(content: frame, completion: .contentProcessed({_ in})) }
        }
    }
    
    private func configurePort(path: String, baud: String) {
        let t = Process(); t.launchPath = "/bin/stty"; t.arguments = ["-f", path, baud, "cs8", "-cstopb", "-parity"]; try? t.run(); t.waitUntilExit()
    }
    
    func stopAll() { shouldRead = false; v2xHandle?.closeFile(); gpsHandle?.closeFile(); listener?.cancel() }
}

@Observableclass V2XNetClient {
    var vehicles: [Int: V2XVehicle] = [:]
    var dangerZones: [Int: V2XDangerZone] = [:]
    var trafficLights: [Int: V2XTrafficLight] = [:]
    var myLocation: CLLocationCoordinate2D?
    var isConnected = false
    
    private var connection: NWConnection?, isRunning = false
    private let serverIP = "192.168.4.1" // Mac- oder Pi-Hotspot IP
    
    func connect() {
        guard connection == nil else { return }
        connection = NWConnection(host: NWEndpoint.Host(serverIP), port: 8080, using: .tcp)
        connection?.stateUpdateHandler = { [weak self] state in
            if state == .ready { self?.isConnected = true; self?.startReceiving() }
            else if case .failed = state { self?.disconnect(); self?.autoReconnect() }
        }
        isRunning = true; connection?.start(queue: .global(qos: .userInteractive))
    }
    
    func disconnect() { isRunning = false; connection?.cancel(); connection = nil; isConnected = false }
    private func autoReconnect() { DispatchQueue.global().asyncAfter(deadline: .now() + 3.0) { [weak self] in self?.connect() } }
    
    private func startReceiving() {
        var remainder = Data()
        func next() {
            guard isRunning, let c = connection else { return }
            c.receive(minimumIncompleteLength: 1, maximumLength: 2048) { [weak self] data, _, complete, _ in
                if let data, !data.isEmpty {
                    remainder.append(data)
                    while let idx = remainder.firstIndex(of: 0x0A) {
                        let line = remainder.subdata(in: 0..<idx); remainder.removeSubrange(0...idx)
                        if let str = String(data: line, encoding: .utf8) { self?.parse(str) }
                    }
                }
                if complete { self?.disconnect() } else if self?.isRunning == true { next() }
            }
        }
        next()
    }
    
    private func parse(_ str: String) {
        guard let data = str.data(using: .utf8),
              let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let type = json["msgType"] as? String, let payload = json["data"] as? [String: Any] else { return }
        
        DispatchQueue.main.async {
            if type == "CAM" {
                guard let id = payload["id"] as? Int, let lat = payload["latitude"] as? Double, let lon = payload["longitude"] as? Double else { return }
                let coord = CLLocationCoordinate2D(latitude: lat, longitude: lon)
                var history = self.vehicles[id]?.history ?? []
                history.append(coord); if history.count > 12 { history.removeFirst() }
                withAnimation(.linear(duration: 0.25)) {
                    self.vehicles[id] = V2XVehicle(id: id, coordinate: coord, heading: payload["heading"] as? Double ?? 0, speed: payload["speed"] as? Double ?? 0, isBraking: payload["isBraking"] as? Bool ?? false, history: history)
                }
            } else if type == "DENM" {
                guard let id = payload["id"] as? Int, let lat = payload["latitude"] as? Double, let lon = payload["longitude"] as? Double else { return }
                self.dangerZones[id] = V2XDangerZone(id: id, type: payload["type"] as? String ?? "Gefahr", coordinate: CLLocationCoordinate2D(latitude: lat, longitude: lon), radiusMeter: payload["radiusMeter"] as? Double ?? 100)
            } else if type == "SPATEM" {
                guard let id = payload["id"] as? Int, let lat = payload["latitude"] as? Double, let lon = payload["longitude"] as? Double else { return }
                self.trafficLights[id] = V2XTrafficLight(id: id, coordinate: CLLocationCoordinate2D(latitude: lat, longitude: lon), currentPhase: payload["currentPhase"] as? String ?? "red", timeToChange: payload["timeToChange"] as? Int ?? 0)
            } else if type == "MY_GPS" {
                guard let lat = payload["latitude"] as? Double, let lon = payload["longitude"] as? Double else { return }
                self.myLocation = CLLocationCoordinate2D(latitude: lat, longitude: lon)
            }
        }
    }
    
    // MATHEMATISCHE GLOSA-BERECHNUNG (Grüne Welle Berater)
    func calculateGlosa(for light: V2XTrafficLight) -> Int? {
        guard let myLoc = myLocation else { return nil }
        let dLat = (light.coordinate.latitude - myLoc.latitude) * .pi / 180
        let dLon = (light.coordinate.longitude - myLoc.longitude) * .pi / 180
        let a = sin(dLat/2)*sin(dLat/2) + cos(myLoc.latitude*.pi/180)*cos(light.coordinate.latitude*.pi/180)*sin(dLon/2)*sin(dLon/2)
        let distance = 6371000.0 * (2 * atan2(sqrt(a), sqrt(1-a)))
        
        if distance > 400 || distance < 10 || light.timeToChange <= 0 { return nil }
        
        let targetSpeed = (distance / Double(light.timeToChange)) * 3.6
        if light.currentPhase == "green" {
            return targetSpeed <= 50 ? Int(targetSpeed) : Int((distance / Double(light.timeToChange + 15)) * 3.6)
        } else {
            return Int(targetSpeed)
        }
    }
}
