struct V2XVehicle: Identifiable, Codable {

    var coordinate: CLLocationCoordinate2D
    var heading: Double
    var speed: Double
    var isBraking: Bool
    var lastSeen: Date
    
    enum CodingKeys: String, CodingKey { case id, heading, speed, isBraking, latitude, longitude }
    
    init(id: Int, coordinate: CLLocationCoordinate2D, heading: Double, speed: Double, isBraking: Bool, lastSeen: Date) {
        self.id = id; self.coordinate = coordinate; self.heading = heading; self.speed = speed; self.isBraking = isBraking; self.lastSeen = lastSeen
    }

        id = try c.decode(Int.self, forKey: .id)
        heading = try c.decode(Double.self, forKey: .heading)
        speed = try c.decode(Double.self, forKey: .speed)
        isBraking = try c.decode(Bool.self, forKey: .isBraking)

        lastSeen = Date()
    }
    
    func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        try c.encode(id, forKey: .id)
        try c.encode(heading, forKey: .heading)
        try c.encode(speed, forKey: .speed)
        try c.encode(isBraking, forKey: .isBraking)

// SLIP Decoder für Pit711-Firmware
class SLIPDecoder {
    static func decode(rawBytes: Data) -> Data {
        var decoded = Data()
        var i = 0
        while i < rawBytes.count {
            let byte = rawBytes[i]
            if byte == 0xDB && i + 1 < rawBytes.count {
                let next = rawBytes[i + 1]
                if next == 0xDC { decoded.append(0xC0) }
                else if next == 0xDD { decoded.append(0xDB) }
                i += 1
            } else { decoded.append(byte) }
            i += 1
        }
        return decoded
    }
}

    func connect() {
        guard connection == nil else { return }
        connection = NWConnection(host: NWEndpoint.Host(serverIP), port: 8080, using: .tcp)
        connection?.stateUpdateHandler = { [weak self] state in
            if state == .ready { self?.isConnected = true; self?.startReceiving() }
            else if case .failed = state { self?.disconnect(); self?.autoReconnect() }
        }
        isRunning = true; connection?.start(queue: .global(qos: .userInteractive))
    }
    
    func disconnect() { isRunning = false; connection?.cancel(); connection = nil; isConnected = false }
    private func autoReconnect() { guard isRunning else { return }; DispatchQueue.global().asyncAfter(deadline: .now() + 3.0) { [weak self] in self?.connect() } }
    
    private func startReceiving() {
        var remainder = Data()
        func next() {
            guard isRunning, let c = connection else { return }
            c.receive(minimumIncompleteLength: 1, maximumLength: 2048) { [weak self] data, _, complete, _ in
                if let data, !data.isEmpty {
                    remainder.append(data)
                    while let idx = remainder.firstIndex(of: 0x0A) {
                        let line = remainder.subdata(in: 0..<idx); remainder.removeSubrange(0...idx)
                        if let str = String(data: line, encoding: .utf8) { self?.parse(str) }
                    }
                }
                if complete { self?.disconnect() } else if self?.isRunning == true { next() }
            }
        }
        next()
    }
    
    private func parse(_ str: String) {
        guard let data = str.data(using: .utf8),
              let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let type = json["msgType"] as? String, let payload = json["data"] as? [String: Any] else { return }
        
        DispatchQueue.main.async {
            if type == "CAM" {
                guard let id = payload["id"] as? Int, let lat = payload["latitude"] as? Double, let lon = payload["longitude"] as? Double else { return }

    func calculateGlosa(for light: V2XTrafficLight) -> Int? {
        guard let myLoc = myLocation else { return nil }
        let dLat = (light.coordinate.latitude - myLoc.latitude) * .pi / 180
        let dLon = (light.coordinate.longitude - myLoc.longitude) * .pi / 180
        let a = sin(dLat/2)*sin(dLat/2) + cos(myLoc.latitude*.pi/180)*cos(light.coordinate.latitude*.pi/180)*sin(dLon/2)*sin(dLon/2)
        let distance = 6371000.0 * (2 * atan2(sqrt(a), sqrt(1-a)))
        
        if distance > 400 || distance < 10 || light.timeToChange <= 0 { return nil }
        
        let targetSpeed = (distance / Double(light.timeToChange)) * 3.6
        if light.currentPhase == "green" {
            return targetSpeed <= 50 ? Int(targetSpeed) : Int((distance / Double(light.timeToChange + 15)) * 3.6)
        } else {
            return Int(targetSpeed)
        }
    }

                    Annotation("Auto", coordinate: v.coordinate) { Image(systemName: "car.up.fill").font(.title3).foregroundColor(v.isBraking ? .red : .blue).rotationEffect(.degrees(v.heading)) }
                }
                
                ForEach(Array(client.trafficLights.values)) { light in
                    Annotation("Ampel", coordinate: light.coordinate) {
                        VStack(spacing: 3) {
                            if let speed = client.calculateGlosa(for: light), speed > 15, speed < 60 {
                                Text("\(speed) km/h").font(.system(size: 9, weight: .bold)).padding(3).background(.green).foregroundColor(.white).cornerRadius(4)
                            }
                            ZStack {
                                Circle().stroke(light.currentPhase == "green" ? Color.green : (light.currentPhase == "yellow" ? Color.yellow : Color.red), lineWidth: 3.5).frame(width: 28, height: 28).background(Circle().fill(.black))
                                Text("\(light.timeToChange)s").font(.system(size: 9, weight: .bold, design: .mono)).foregroundColor(.white)
                            }
                        }
                    }
                }
            }.mapStyle(.standard(showsTraffic: false))
            
            HStack {
                Circle().fill(client.isConnected ? .green : .red).frame(width: 8, height: 8)
                Text(client.isConnected ? "C-ITS Live Cockpit" : "Suche Mac-Zentrale...").font(.caption).bold()
                Spacer()
                Text("Targets: \(client.vehicles.count)").font(.caption2).foregroundColor(.gray)
            }.padding().background(Color(.systemBackground).opacity(0.9)).cornerRadius(10).padding()
        }.onAppear { client.connect() }
    }
